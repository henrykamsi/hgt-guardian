class GlobalHGTBridge(private val activity: Activity, private val viewType: String) {
    @JavascriptInterface
    fun postMessage(status: String) {
        if (status == "VERIFIED_ALL") {
            activity.runOnUiThread {
                when(viewType) {
                    "GATEKEEPER" -> showMainAppUI()
                    "LOGIN" -> showLoginUI()
                    "SIGNUP" -> showSignupUI()

            }
        }
    }

iOS Swift Integration

These snippets inject the HGT Guardian JS link via WKWebView to secure your native Apple iOS screens.

1. First Redirect (Gatekeeper)

SceneDelegate.swift
let htmlString = "<html><head><script src='https://henrykamsi.github.io/hgt-guardian/hgt-guardian.js'></script></head><body><script>HGT_Guardian.init({ onSuccess: () => window.webkit.messageHandlers.HGTBridge.postMessage('VERIFIED') });</script></body></html>"
webView.loadHTMLString(htmlString, baseURL: nil)

func userContentController(_ userContentController: WKUserContentController, didReceive message: WKScriptMessage) {
    if message.name == "HGTBridge", let status = message.body as? String, status == "VERIFIED" {
        self.transitionToMainApp()
    }
}

4. Master Code (Combination)

let htmlString = "<html><head><script src='https://henrykamsi.github.io/hgt-guardian/hgt-guardian.js'></script></head><body><script>HGT_Guardian.init({ onSuccess: () => window.webkit.messageHandlers.HGTBridge.postMessage('VERIFIED_ALL') });</script></body></html>"

extension UIViewController: WKScriptMessageHandler {
    public func userContentController(_ userContentController: WKUserContentController, didReceive message: WKScriptMessage) {
        guard message.name == "HGTBridge", let status = message.body as? String, status == "VERIFIED_ALL" else { return }
        
        DispatchQueue.main.async {
            if self is LoginViewController { (self as! LoginViewController).unlockForm() }
            else if self is SignupViewController { (self as! SignupViewController).unlockForm() }
            else { self.dismiss(animated: true) }
        }
    }
}
